#include <SPI.h>
#include <RF24.h>
#include <SoftwareSerial.h>

/* ===== NRF24 ===== */
#define CE_PIN 10
#define CSN_PIN 9
RF24 radio(CE_PIN, CSN_PIN);
uint8_t address[][6] = {"1Node", "2Node"};

/* ===== VOZ ===== */
SoftwareSerial voz(A4, A5); // RX, TX

/* ===== BOTONES ===== */
#define BTN_ADELANTE   2
#define BTN_ATRAS      3
#define BTN_IZQUIERDA  4
#define BTN_DERECHA    5
#define BTN_VOZ        6

uint8_t payload = 6;        // PARAR
uint8_t ultimoEnviado = 255;

void setup() {
  Serial.begin(115200);
  voz.begin(9600);

  pinMode(BTN_ADELANTE, INPUT_PULLUP);
  pinMode(BTN_ATRAS, INPUT_PULLUP);
  pinMode(BTN_IZQUIERDA, INPUT_PULLUP);
  pinMode(BTN_DERECHA, INPUT_PULLUP);

  radio.begin();
  radio.setAutoAck(false);
  radio.setDataRate(RF24_250KBPS);
  radio.setPALevel(RF24_PA_LOW);
  radio.setChannel(76);

  radio.openWritingPipe(address[0]);
  radio.stopListening();

  Serial.println("=== EMISOR LISTO ===");
}

void loop() {
  payload = 15; // PARAR por defecto

  /* ===== VOZ (prioridad) ===== */
  if (voz.available()) {
    uint8_t id = voz.read();
    if (id >= 6 && id <= 10) {
      payload = id;
      Serial.print("VOZ: ");
      Serial.println(payload);
    }
  }
  /* ===== BOTONES ===== */
  else {
    if (!digitalRead(BTN_ADELANTE) && !digitalRead(BTN_DERECHA)) payload = 1;
    else if (!digitalRead(BTN_ADELANTE) && !digitalRead(BTN_IZQUIERDA)) payload = 2;
    else if (!digitalRead(BTN_ATRAS) && !digitalRead(BTN_DERECHA)) payload = 3;
    else if (!digitalRead(BTN_ATRAS) && !digitalRead(BTN_IZQUIERDA)) payload = 4;
    else if (!digitalRead(BTN_ADELANTE)) payload = 5;
    else if (!digitalRead(BTN_ATRAS)) payload = 11;
    else if (!digitalRead(BTN_DERECHA)) payload = 14;
    else if (!digitalRead(BTN_IZQUIERDA)) payload = 13;
    else if (!digitalRead(BTN_VOZ)) payload = 12;
  }

  /* ===== ENVÍO ===== */
  if (payload != ultimoEnviado) {
    radio.write(&payload, sizeof(payload));
    Serial.print("TX: ");
    Serial.println(payload);
    ultimoEnviado = payload;
  }

  delay(50);
}
